% The network as a standalone figure, in the type and palette of scripts/style.py.
% Build with XeLaTeX (Inter is vendored in ../../fonts): xelatex architecture.tex
\documentclass[border=1pt]{standalone}
\usepackage{amsmath}
\usepackage{fontspec}
\setmainfont{Inter}[Path=../../fonts/, Extension=.ttf, UprightFont=*-Regular,
  ItalicFont=*-Italic, BoldFont=*-SemiBold]
\usepackage[italic]{mathastext}
\usepackage{tikz}
\usetikzlibrary{positioning,fit,arrows.meta,calc,backgrounds}

% style.py: ink, chrome and the categorical slots
\definecolor{ink}{HTML}{16161A}
\definecolor{ink2}{HTML}{55555C}
\definecolor{muted}{HTML}{909098}
\definecolor{axis}{HTML}{B8B8BF}
\definecolor{wash}{HTML}{F2F2F4}
\definecolor{blue}{HTML}{2A78D6}
\definecolor{orange}{HTML}{EB6834}
\definecolor{violet}{HTML}{6250D6}
% the source stream blue, the data stream neutral, sinks violet, conditioning orange
\colorlet{xfill}{blue!12}
\colorlet{xchip}{blue!6}
\colorlet{yfill}{wash}
\colorlet{sfill}{violet!11}
\colorlet{cfill}{orange!9}

\newcommand{\sm}{\fontsize{6.3}{7.2}\selectfont}
\begin{document}
\fontsize{7}{8.4}\selectfont
\hyphenpenalty=10000
\begin{tikzpicture}[
    x=1mm, y=1mm, text=ink,
    >={Stealth[length=2.4pt,width=2.2pt]},
    box/.style={draw=#1, rounded corners=2pt, line width=0.55pt, align=center,
                inner sep=1.8pt, text width=33mm},
    xbox/.style={box=blue, fill=xfill},
    ybox/.style={box=muted, fill=yfill},
    sbox/.style={box=violet, fill=sfill},
    cbox/.style={box=orange, fill=cfill},
    chip/.style={draw=#1, rounded corners=1.5pt, line width=0.45pt, align=center,
                 inner sep=0.8pt, minimum height=7.5mm},
    xchip/.style={chip=blue, fill=xchip},
    ychip/.style={chip=muted, fill=white},
    add/.style={circle, draw=#1, line width=0.45pt, fill=white, inner sep=0pt,
                minimum size=3.4mm},
    small/.style={font=\fontsize{6.3}{7.4}\selectfont},
    note/.style={small, text=ink2, align=center},
    title/.style={font=\fontsize{7}{8.4}\selectfont\bfseries, text=ink},
    arr/.style={->, line width=0.55pt, draw=ink2},
    res/.style={->, line width=0.45pt, draw=#1, rounded corners=2pt},
    cond/.style={->, line width=0.5pt, draw=orange, dash pattern=on 2pt off 1.2pt},
    kv/.style={->, line width=1pt, draw=ink2},
  ]
  \def\ys{40}
  \def\yd{0}

  % ----------------------------------------------------------------- inputs and embeddings
  \node[xbox, anchor=west] (xemb) at (12, \ys)
    {\textbf{source tokens}\\ PLE of coordinates\\ + slot + Fourier($f_0$ bin)\\
     + PLE($\hat X_1$), self-conditioning};
  \node[sbox, anchor=west] (semb) at (12, 28.4) {\textbf{sink tokens}: $\nu$ + learned};
  \node[cbox, anchor=west] (cemb) at (12, 18.4)
    {\textbf{conditioning}\\ $\nu$ from $t$, $\log_{10}(1-t)$, $\ln f_w$\\ $\nu_f$ from $\ln f_w$};
  \node[ybox, anchor=west] (yemb) at (12, \yd)
    {\textbf{data tokens}\\ spectra of $Q$ bins per token\\ PLE of amplitude and phase\\
     + Fourier(centre bin)};
  \node[note, anchor=east, align=right] (xin) at (9, \ys) {$X_t$\\ $S\times 8$};
  \node[note, anchor=east, align=right] (cin) at (9, 18.4) {$t$, $f_w$};
  \node[note, anchor=east, align=right] (yin) at (9, \yd) {$y$\\ $K\times 6$};
  \draw[arr] (xin) -- (xemb);
  \draw[arr] (cin) -- (cemb);
  \draw[arr] (yin) -- (yemb);
  \node[title, anchor=south] at (xemb.north) {embeddings};

  % ----------------------------------------------------------------- one block, two streams
  \foreach \s/\y/\c/\what in {x/\ys/blue/{sources, sinks, data}, y/\yd/muted/{data only}} {
    \node[\s chip, minimum width=9mm] (\s a1) at (66, \y) {adaLN};
    \node[\s chip, text width=22mm, right=2.4mm of \s a1] (\s att)
      {attention\\[-0.6pt] {\sm\color{ink2}\what}};
    \node[add=\c, right=2.2mm of \s att] (\s p1) {$+$};
    \node[\s chip, minimum width=9mm, right=2.2mm of \s p1] (\s a2) {adaLN};
    \node[\s chip, minimum width=11mm, right=2.4mm of \s a2] (\s ff) {SwiGLU};
    \node[add=\c, right=2.2mm of \s ff] (\s p2) {$+$};
    \foreach \a/\b in {a1/att, att/p1, p1/a2, a2/ff, ff/p2} \draw[arr] (\s\a) -- (\s\b);
  }

  % residual paths: above the source stream, below the data stream
  \coordinate (xin1) at ($(xa1.west)+(-3,0)$);
  \coordinate (yin1) at ($(ya1.west)+(-3,0)$);
  \draw[res=blue] (xin1) |- ($(xatt.north)+(0,2.6)$) -| (xp1.north);
  \draw[res=blue] ($(xp1.east)+(1,0)$) |- ($(xff.north)+(0,2.6)$) -| (xp2.north);
  \draw[res=muted] (yin1) |- ($(yatt.south)+(0,-2.6)$) -| (yp1.south);
  \draw[res=muted] ($(yp1.east)+(1,0)$) |- ($(yff.south)+(0,-2.6)$) -| (yp2.south);

  % one-way attention: data keys and values reach the sources, never the reverse
  \draw[kv] (yatt.north) -- node[note, fill=white, inner sep=1pt, pos=0.5]
    {keys, values\\ of the data} (yatt.north |- xatt.south);

  % conditioning: nu modulates the source stream, sinks and head; nu_f the data stream
  \def\bx{33.4}
  \def\by{6.6}
  \draw[cond, -] ($(cemb.east)+(0,1.6)$) -| (55.5, \bx) -- (xp2.east |- 0,\bx);
  \draw[cond, -] ($(cemb.east)+(0,-1.6)$) -| (57.5, \by) -- (yp2.east |- 0,\by);
  \draw[cond] (55.5, \bx) |- ($(semb.east)+(0,0)$);
  \foreach \n in {xa1, xa2, xp1, xp2} \draw[cond] (\n.center |- 0,\bx) -- (\n.south);
  \foreach \n in {ya1, ya2, yp1, yp2} \draw[cond] (\n.center |- 0,\by) -- (\n.north);
  \node[small, text=orange, anchor=south west, inner sep=0.6pt] at (58.5, \bx) {$\nu$};
  \node[small, text=orange, anchor=north west, inner sep=0.6pt] at (60.5, \by) {$\nu_f$};

  % RoPE on one frequency axis, in both attentions
  \node[note, anchor=west, align=left] at ($(xp1.west |- 0,20)+(1,0)$)
    {rotary positions on one\\ frequency axis: a source at\\ its $f_0$ bin, a data token\\ at its centre bin};

  % embeddings into the block; the sinks join the source stream
  \draw[arr] (xemb.east) -- (xa1.west);
  \draw[arr] ($(semb.east)+(0,1.2)$) -| ($(xin1)+(-1.4,0)$);
  \draw[arr] (yemb.east) -- (ya1.west);

  % block frame, behind everything
  \begin{scope}[on background layer]
    \node[draw=axis, line width=0.5pt, rounded corners=3pt, fill=white,
          fit={(52,-12.5) ($(xp2.east)+(0,11)$)}, inner sep=1mm] (frame) {};
  \end{scope}
  \node[title, anchor=north west] at ($(frame.north west)+(1.2,-0.8)$)
    {source stream \textcolor{ink2}{\normalfont per draw and step}};
  \node[title, anchor=south west] at ($(frame.south west)+(1.2,0.8)$)
    {data stream \textcolor{ink2}{\normalfont once per observation}};
  \node[title, anchor=north east] at ($(frame.north east)+(-1.2,-0.8)$) {block $\times N$};

  % ----------------------------------------------------------------- head and cache
  \node[xchip, text width=18mm, anchor=west] (head) at ($(frame.east |- 0,\ys)+(3.5,0)$)
    {head\\[-0.6pt] {\sm\color{ink2}adaLN, SwiGLU}};
  \draw[arr] (xp2) -- (head);
  \node[anchor=west, inner sep=1pt, align=center] (out) at ($(head.east)+(2.4,0)$)
    {$\hat X_1$\\ {\sm\color{ink2}$S\times 8$}};
  \draw[arr] (head) -- (out);
  \draw[cond] (xp2.east |- 0,\bx) -| (head.south);
  \node[ychip, text width=18mm, anchor=west] (cache) at ($(frame.east |- 0,\yd)+(3.5,0)$)
    {cache\\[-0.6pt] {\sm\color{ink2}keys and values\\[-1pt] of all blocks}};
  \draw[arr] (yp2) -- (cache);
\end{tikzpicture}
\end{document}
